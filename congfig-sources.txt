1. Terrafrom intall
    wget -O - https://apt.releases.hashicorp.com/gpg | sudo gpg --dearmor -o /usr/share/keyrings/hashicorp-archive-keyring.gpg
    echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/hashicorp-archive-keyring.gpg] https://apt.releases.hashicorp.com $(grep -oP '(?<=UBUNTU_CODENAME=).*' /etc/os-release || lsb_release -cs) main" | sudo tee /etc/apt/sources.list.d/hashicorp.list
    sudo apt update && sudo apt install terraform
2. Cấu hình AWS CLI trên WSL
    # Cài AWS CLI (trên WSL Ubuntu)
    curl "https://awscli.amazonaws.com/awscli-exe-linux-x86_64.zip" -o "awscliv2.zip"
    unzip awscliv2.zip
    sudo ./aws/install

    # Kiểm tra
    aws --version

    # Cấu hình credential
    aws configure

4. Bật Auto-pause cho Redshift Serverless
    AWS Terraform provider hiện chưa support auto-pause qua resource (tính đến provider v5). Phải bật bằng CLI sau khi apply:
    aws redshift-serverless update-workgroup \
  --workgroup-name ecommerce-de-pipeline-wg \
  --base-capacity 8 \
  --config-parameters parameterKey=autoPause,parameterValue=true \
                     parameterKey=autoPauseDelay,parameterValue=300
